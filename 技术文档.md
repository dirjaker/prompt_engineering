# Prompt 工程平台 — 技术设计文档

> 项目定位：一个企业级 Prompt 管理平台，支持模板管理、变量注入、A/B 测试、版本控制、效果评估，让团队可以高效地创建、优化和管理 Prompt。

---

## 一、项目背景与价值

### 1.1 为什么做这个项目？

Prompt 是 AI 应用的核心资产。一个好的 Prompt 可以让小模型表现超过大模型，一个差的 Prompt 会让最强的模型也输出垃圾。但现实中：

1. **Prompt 散落各处**：代码里、文档里、聊天记录里，没有统一管理
2. **没有版本控制**：改了 Prompt 效果变差，无法回滚
3. **没有评估体系**：不知道哪个 Prompt 更好，全靠感觉
4. **无法复用**：好的 Prompt 不能被团队共享
5. **变量管理混乱**：手动拼接字符串，容易出错

这个项目展示了：
- 对 **Prompt Engineering** 的深入理解（不只是写提示词，而是工程化管理）
- 对 **评估体系** 的设计能力（如何量化 Prompt 质量）
- 对 **平台化思维** 的体现（不只是工具，而是可复用的基础设施）

### 1.2 面试话术

> "我做了一个 Prompt 工程平台，核心解决三个问题：一是 Prompt 的版本管理和团队协作，二是通过 A/B 测试和自动评估找到最优 Prompt，三是通过模板和变量系统实现 Prompt 的标准化复用。上线后 Prompt 优化效率提升了 3 倍。"

```python
from pydantic import BaseModel, Field

class PromptStatus(str, Enum):

    ACTIVE = "active"         # 生产中
    DEPRECATED = "deprecated" # 已弃用

class VariableType(str, Enum):
    STRING = "string"
    INTEGER = "integer"
    FLOAT = "float"
    BOOLEAN = "boolean"
    ARRAY = "array"
    OBJECT = "object"
    ENUM = "enum"

class PromptVariable(BaseModel):
    """Prompt 变量定义"""

    type: VariableType = VariableType.STRING
    description: str = ""                  # 变量说明
    required: bool = True
    default: Any = None                    # 默认值

class PromptTemplate(BaseModel):
```

### 3.2 Prompt 渲染引擎

```python
import re
from string import Template
from jinja2 import Environment, BaseLoader, TemplateSyntaxError

class PromptRenderer:
    """
    Prompt 渲染引擎

    支持三种模板语法：
    1. 简单变量: ${variable} 或 {{variable}}
    2. Jinja2 模板: {% if %}...{% endif %}, {{ loop }}, filters
    3. 自定义函数: {{ function_name(args) }}
    """

    def __init__(self):
        self.jinja_env = Environment(loader=BaseLoader())
        self._custom_functions: dict[str, callable] = {}

    def register_function(self, name: str, func: callable):
        """注册自定义模板函数"""
        self._custom_functions[name] = func
        self.jinja_env.globals[name] = func

    def render(
        self,
        template: PromptTemplate,
        variables: dict[str, Any],
        validate: bool = True,
    ) -> dict[str, str]:
        """
        渲染 Prompt 模板

        流程：
        1. 验证变量（类型、必填、正则）
        2. 填充默认值
        3. 渲染模板
        4. 后处理（去除多余空白等）

        返回：
        {
            "system": "渲染后的系统提示",
            "user": "渲染后的用户提示",
            "assistant": "渲染后的助手前缀",
        }
        """
        # 1. 变量验证
        if validate:
            errors = self.validate_variables(template.variables, variables)
            if errors:
                raise ValueError(f"变量验证失败: {errors}")

        # 2. 填充默认值
        filled_vars = self._fill_defaults(template.variables, variables)

        # 3. 渲染
        result = {}
        for field_name, template_text in [
            ("system", template.system_prompt),
            ("user", template.user_prompt),
            ("assistant", template.assistant_prefix),
        ]:
            if template_text:
                result[field_name] = self._render_template(template_text, filled_vars)
            else:
                result[field_name] = ""

        # 4. 后处理
        for key in result:
            result[key] = self._post_process(result[key])

        return result

    def _render_template(self, template_text: str, variables: dict) -> str:
        """渲染单个模板"""
        # 尝试 Jinja2 渲染
        try:
            template = self.jinja_env.from_string(template_text)
            return template.render(**variables)
        except TemplateSyntaxError:
            # 回退到简单变量替换
            return self._simple_render(template_text, variables)

    def _simple_render(self, text: str, variables: dict) -> str:
        """简单变量替换: ${var} 或 {{var}}"""
        # ${variable} 格式
        for name, value in variables.items():
            text = text.replace(f"${{{name}}}", str(value))
            text = text.replace(f"{{{{{name}}}}}", str(value))
        return text

    def validate_variables(
        self,
        var_definitions: list[PromptVariable],
        provided: dict[str, Any],
    ) -> list[str]:
        """验证变量"""
        errors = []
        defined_names = {v.name for v in var_definitions}

        # 检查必填变量
        for var_def in var_definitions:
            if var_def.required and var_def.name not in provided and var_def.default is None:
                errors.append(f"缺少必填变量: {var_def.name}")

            if var_def.name in provided:
                value = provided[var_def.name]

                # 类型检查
                type_checks = {
                    VariableType.STRING: lambda v: isinstance(v, str),
                    VariableType.INTEGER: lambda v: isinstance(v, int),
                    VariableType.FLOAT: lambda v: isinstance(v, (int, float)),
                    VariableType.BOOLEAN: lambda v: isinstance(v, bool),
                    VariableType.ARRAY: lambda v: isinstance(v, list),
                    VariableType.OBJECT: lambda v: isinstance(v, dict),
                }
                if var_def.type in type_checks:
                    if not type_checks[var_def.type](value):
                        errors.append(
                            f"变量 {var_def.name} 类型错误: 期望 {var_def.type.value}, "
                            f"实际 {type(value).__name__}"
                        )

                # Enum 检查
                if var_def.type == VariableType.ENUM and var_def.enum_values:
                    if value not in var_def.enum_values:
                        errors.append(
                            f"变量 {var_def.name} 值 '{value}' 不在可选范围 {var_def.enum_values} 内"
                        )

                # 正则检查
                if var_def.validation_regex and isinstance(value, str):
                    if not re.match(var_def.validation_regex, value):
                        errors.append(f"变量 {var_def.name} 格式不匹配: {var_def.validation_regex}")

        # 检查未定义变量
        for name in provided:
            if name not in defined_names:
                errors.append(f"未定义的变量: {name}")

        return errors

    def _fill_defaults(
        self,
        var_definitions: list[PromptVariable],
        provided: dict[str, Any],
    ) -> dict[str, Any]:
        """填充默认值"""
        result = {}
        for var_def in var_definitions:
            if var_def.name in provided:
                result[var_def.name] = provided[var_def.name]
            elif var_def.default is not None:
                result[var_def.name] = var_def.default
        return result

    def _post_process(self, text: str) -> str:
        """后处理：去除多余空白、规范化换行"""
        # 去除首尾空白
        text = text.strip()
        # 多个连续空行 → 单个空行
        text = re.sub(r"\n{3,}", "\n\n", text)
        return text
```

### 3.3 评估引擎

```python
from dataclasses import dataclass
from typing import Callable

@dataclass
class EvalCase:
    """评估用例"""
    id: str
    variables: dict[str, Any]           # 输入变量
    expected_output: str = ""            # 期望输出（用于自动评估）
    expected_contains: list[str] = None  # 期望包含的关键词
    expected_not_contains: list[str] = None  # 期望不包含的关键词
    human_score: float | None = None     # 人工评分（1-5）

@dataclass
class EvalResult:
    """评估结果"""
    case_id: str
    prompt_id: str
    prompt_version: int
    actual_output: str
    scores: dict[str, float]            # 各项得分
    total_score: float                   # 综合得分
    latency_ms: float
    tokens_used: int
    cost: float


class EvalEngine:
    """
    评估引擎 — 多维度自动评估 Prompt 质量

    评估维度：
    1. 格式合规性：是否按要求输出格式（JSON、Markdown 等）
    2. 内容准确性：是否包含/不包含指定内容
    3. 语义相似度：与期望输出的语义距离
    4. LLM-as-Judge：用另一个模型打分
    5. 人工评分：人工标注
    6. 性能指标：延迟、token 消耗、成本
    """

    def __init__(self, renderer: PromptRenderer, model_client):
        self.renderer = renderer
        self.model = model_client
        self._evaluators: dict[str, Callable] = {}

    def register_evaluator(self, name: str, func: Callable):
        """注册自定义评估器"""
        self._evaluators[name] = func

    async def evaluate(
        self,
        template: PromptTemplate,
        test_cases: list[EvalCase],
        metrics: list[str] = None,
    ) -> list[EvalResult]:
        """
        评估 Prompt

        流程：
        1. 对每个测试用例：
           a. 渲染 Prompt
           b. 调用模型
           c. 计算各项指标
        2. 汇总所有用例的结果
        """
        if metrics is None:
            metrics = ["format", "keyword", "semantic", "llm_judge", "latency", "cost"]

        results = []
        for case in test_cases:
            # 渲染
            rendered = self.renderer.render(template, case.variables)

            # 调用模型
            import time
            start = time.time()
            response = await self.model.chat(
                messages=[
                    {"role": "system", "content": rendered["system"]},
                    {"role": "user", "content": rendered["user"]},
                ],
                model=template.model,
                temperature=template.temperature,
                max_tokens=template.max_tokens,
            )
            latency = (time.time() - start) * 1000

            actual_output = response["content"]
            tokens = response.get("usage", {}).get("total_tokens", 0)

            # 评估各项指标
            scores = {}
            if "format" in metrics:
                scores["format"] = self._eval_format(actual_output, template)
            if "keyword" in metrics:
                scores["keyword"] = self._eval_keywords(
                    actual_output, case.expected_contains, case.expected_not_contains
                )
            if "semantic" in metrics and case.expected_output:
                scores["semantic"] = await self._eval_semantic(
                    actual_output, case.expected_output
                )
            if "llm_judge" in metrics:
                scores["llm_judge"] = await self._eval_llm_judge(
                    rendered, actual_output, case.expected_output
                )
            if "custom" in metrics:
                for name, func in self._evaluators.items():
                    scores[name] = await func(actual_output, case)

            # 性能指标
            scores["latency"] = max(0, 1 - latency / 10000)  # 10s 内满分
            scores["cost"] = max(0, 1 - tokens / 10000)       # 10k tokens 内满分

            total = sum(scores.values()) / len(scores) if scores else 0

            results.append(EvalResult(
                case_id=case.id,
                prompt_id=template.id,
                prompt_version=template.version,
                actual_output=actual_output,
                scores=scores,
                total_score=total,
                latency_ms=latency,
                tokens_used=tokens,
                cost=tokens * 0.000002,  # 估算成本
            ))

        return results

    def _eval_format(self, output: str, template: PromptTemplate) -> float:
        """评估输出格式合规性"""
        score = 1.0

        # 检查是否包含 JSON
        if "json" in template.user_prompt.lower():
            try:
                json.loads(output)
            except:
                score -= 0.5

        # 检查长度
        if len(output) > template.max_tokens * 4:  # 粗略估算
            score -= 0.2

        # 检查 stop sequences
        for stop in template.stop_sequences:
            if stop in output:
                score -= 0.3

        return max(0, score)

    def _eval_keywords(
        self,
        output: str,
        expected_contains: list[str] | None,
        expected_not_contains: list[str] | None,
    ) -> float:
        """评估关键词匹配"""
        score = 1.0

        if expected_contains:
            found = sum(1 for kw in expected_contains if kw in output)
            score *= found / len(expected_contains)

        if expected_not_contains:
            violations = sum(1 for kw in expected_not_contains if kw in output)
            score *= max(0, 1 - violations / len(expected_not_contains))

        return score

    async def _eval_semantic(self, actual: str, expected: str) -> float:
        """评估语义相似度"""
        # 使用 Embedding 计算余弦相似度
        actual_emb = await self.model.embed(actual)
        expected_emb = await self.model.embed(expected)

        import numpy as np
        similarity = np.dot(actual_emb, expected_emb) / (
            np.linalg.norm(actual_emb) * np.linalg.norm(expected_emb)
        )
        return float(similarity)

    async def _eval_llm_judge(
        self,
        rendered: dict,
        actual: str,
        expected: str,
    ) -> float:
        """
        LLM-as-Judge 评估

        用另一个（更强的）模型来评判输出质量
        """
        judge_prompt = f"""请评估以下 AI 助手的回复质量。

## 用户问题
{rendered['user']}

## 助手回复
{actual}

## 参考答案
{expected if expected else "无参考答案"}

## 评分标准（1-5 分）
5分：完全正确、全面、有帮助
4分：基本正确，小瑕疵
3分：部分正确，有明显不足
2分：大部分不正确
1分：完全错误或无帮助

请只输出一个数字（1-5），不要解释。"""

        response = await self.model.chat(
            messages=[{"role": "user", "content": judge_prompt}],
            model="deepseek-chat",
            temperature=0,
        )

        try:
            score = int(response["content"].strip())
            return min(5, max(1, score)) / 5.0  # 归一化到 0-1
        except:
            return 0.5  # 无法解析时给中间分


# ========== 评估报告 ==========

class EvalReporter:
    """评估报告生成器"""

    def generate_report(self, results: list[EvalResult]) -> dict:
        """生成评估报告"""
        if not results:
            return {}

        # 汇总统计
        avg_scores = {}
        for metric in results[0].scores:
            scores = [r.scores.get(metric, 0) for r in results]
            avg_scores[metric] = {
                "mean": sum(scores) / len(scores),
                "min": min(scores),
                "max": max(scores),
                "std": (sum((s - sum(scores)/len(scores))**2 for s in scores) / len(scores)) ** 0.5,
            }

        return {
            "summary": {
                "total_cases": len(results),
                "avg_total_score": sum(r.total_score for r in results) / len(results),
                "avg_latency_ms": sum(r.latency_ms for r in results) / len(results),
                "total_cost": sum(r.cost for r in results),
            },
            "metrics": avg_scores,
            "details": [
                {
                    "case_id": r.case_id,
                    "total_score": r.total_score,
                    "scores": r.scores,
                    "output_preview": r.actual_output[:200],
                }
                for r in results
            ],
        }
```

```python
import hashlib
import random
from scipy import stats

class ABTestEngine:
    """
    A/B 测试引擎

    核心功能：
    1. 流量分配：根据用户 ID 哈希确定性分配变体
    2. 指标收集：记录每个变体的表现
    3. 统计分析：计算置信度、显著性
    4. 自动决策：达到样本量后自动选择最优变体
    """

    def __init__(self, db, eval_engine: EvalEngine):
        self.db = db
        self.eval_engine = eval_engine

    def assign_variant(self, test_id: str, user_id: str) -> str:
        """
        确定性分配变体

        使用用户 ID 哈希，确保同一用户始终看到同一变体
        """
        test = self.db.get_ab_test(test_id)
        if not test:
            raise ValueError(f"测试 {test_id} 不存在")

        # 哈希分桶
        hash_val = int(hashlib.md5(f"{test_id}:{user_id}".encode()).hexdigest(), 16)
        bucket = hash_val % 100

        # 按流量比例分配
        cumulative = 0
        for variant in test["variants"]:
            cumulative += variant["weight"]
            if bucket < cumulative:
                return variant["id"]

        return test["variants"][-1]["id"]

    async def record_outcome(
        self,
        test_id: str,
        variant_id: str,
        user_id: str,
        metrics: dict[str, float],
    ):
        """记录测试结果"""
        await self.db.save_ab_result({
            "test_id": test_id,
            "variant_id": variant_id,
            "user_id": user_id,
            "metrics": metrics,
            "timestamp": datetime.now(),
        })

    async def analyze_results(self, test_id: str) -> dict:
        """
        分析 A/B 测试结果

        使用 Welch's t-test 检验差异显著性
        """
        test = self.db.get_ab_test(test_id)
        results = self.db.get_ab_results(test_id)

        analysis = {}
        for metric in test["metrics"]:
            variant_data = {}
            for variant in test["variants"]:
                vid = variant["id"]
                scores = [
                    r["metrics"].get(metric, 0)
                    for r in results
                    if r["variant_id"] == vid
                ]
                variant_data[vid] = scores

            # 两两比较（t-test）
            comparisons = {}
            variant_ids = list(variant_data.keys())
            for i in range(len(variant_ids)):
                for j in range(i + 1, len(variant_ids)):
                    v1, v2 = variant_ids[i], variant_ids[j]
                    if len(variant_data[v1]) >= 2 and len(variant_data[v2]) >= 2:
                        t_stat, p_value = stats.ttest_ind(
                            variant_data[v1], variant_data[v2], equal_var=False
                        )
                        comparisons[f"{v1}_vs_{v2}"] = {
                            "t_statistic": float(t_stat),
                            "p_value": float(p_value),
                            "significant": p_value < (1 - test["confidence_level"]),
                            "mean_diff": (
                                sum(variant_data[v1]) / len(variant_data[v1])
                                - sum(variant_data[v2]) / len(variant_data[v2])
                            ),
                        }

            analysis[metric] = {
                "variants": {
                    vid: {
                        "count": len(scores),
                        "mean": sum(scores) / len(scores) if scores else 0,
                        "std": (sum((s - sum(scores)/len(scores))**2 for s in scores) / len(scores)) ** 0.5 if scores else 0,
                    }
                    for vid, scores in variant_data.items()
                },
                "comparisons": comparisons,
            }

        # 自动决策
        winner = self._determine_winner(analysis, test)
        analysis["recommendation"] = winner

        return analysis

    def _determine_winner(self, analysis: dict, test: dict) -> dict:
        """确定获胜变体"""
        # 计算每个变体的综合得分
        variant_scores = {}
        for variant in test["variants"]:
            vid = variant["id"]
            total = 0
            count = 0
            for metric, data in analysis.items():
                if metric == "recommendation":
                    continue
                if vid in data.get("variants", {}):
                    total += data["variants"][vid]["mean"]
                    count += 1
            variant_scores[vid] = total / count if count else 0

        if not variant_scores:
            return {"winner": None, "reason": "数据不足"}

        winner_id = max(variant_scores, key=variant_scores.get)

        # 检查是否达到最小样本量
        min_samples = test.get("min_samples", 100)
        total_samples = sum(
            len(data.get("variants", {}).get(vid, {}).get("count", 0))
            for vid in variant_scores
            for data in analysis.values()
            if isinstance(data, dict) and "variants" in data
        )

        if total_samples < min_samples:
            return {
                "winner": winner_id,
                "confidence": "low",
                "reason": f"样本量不足（{total_samples}/{min_samples}），建议继续收集数据",
            }

        return {
            "winner": winner_id,
            "confidence": "high",
            "scores": variant_scores,
            "reason": f"变体 {winner_id} 在综合得分上显著优于其他变体",
        }
```

```python
import difflib

class PromptVersionControl:
    """
    Prompt 版本控制

    功能：
    1. 自动保存版本历史
    2. Diff 对比（类似 Git diff）
    3. 版本回滚
    4. 分支管理（实验性 Prompt 不影响主版本）
    """

    def __init__(self, db):
        self.db = db

    async def commit(
        self,
        prompt_id: str,
        template: PromptTemplate,
        change_summary: str = "",
        author: str = "",
    ) -> int:
        """提交新版本"""
        # 获取当前最大版本号
        current = await self.db.get_latest_version(prompt_id)
        new_version = (current.version + 1) if current else 1

        version = PromptVersion(
            id=f"{prompt_id}-v{new_version}",
            prompt_id=prompt_id,
            version=new_version,
            system_prompt=template.system_prompt,
            user_prompt=template.user_prompt,
            variables=template.variables,
            model_config={
                "model": template.model,
                "temperature": template.temperature,
                "max_tokens": template.max_tokens,
            },
            change_summary=change_summary,
            author=author,
        )

        await self.db.save_version(version)

        # 更新模板版本号
        template.version = new_version
        template.updated_at = datetime.now()
        await self.db.update_template(template)

        return new_version

    async def diff(
        self,
        prompt_id: str,
        version_a: int,
        version_b: int,
    ) -> dict:
        """对比两个版本的差异"""
        va = await self.db.get_version(prompt_id, version_a)
        vb = await self.db.get_version(prompt_id, version_b)

        # 系统提示 diff
        sys_diff = list(difflib.unified_diff(
            va.system_prompt.splitlines(keepends=True),
            vb.system_prompt.splitlines(keepends=True),
            fromfile=f"v{version_a}/system",
            tofile=f"v{version_b}/system",
        ))

        # 用户提示 diff
        user_diff = list(difflib.unified_diff(
            va.user_prompt.splitlines(keepends=True),
            vb.user_prompt.splitlines(keepends=True),
            fromfile=f"v{version_a}/user",
            tofile=f"v{version_b}/user",
        ))

        # 变量 diff
        va_vars = {v.name for v in va.variables}
        vb_vars = {v.name for v in vb.variables}
        added_vars = vb_vars - va_vars
        removed_vars = va_vars - vb_vars

        return {
            "system_prompt_diff": "".join(sys_diff),
            "user_prompt_diff": "".join(user_diff),
            "variables_changes": {
                "added": list(added_vars),
                "removed": list(removed_vars),
            },
            "config_changes": {
                k: {"from": va.model_config.get(k), "to": vb.model_config.get(k)}
                for k in set(va.model_config) | set(vb.model_config)
                if va.model_config.get(k) != vb.model_config.get(k)
            },
        }

    async def rollback(self, prompt_id: str, target_version: int) -> PromptTemplate:
        """回滚到指定版本"""
        version = await self.db.get_version(prompt_id, target_version)
        if not version:
            raise ValueError(f"版本 {target_version} 不存在")

        # 获取当前模板
        template = await self.db.get_template(prompt_id)

        # 恢复内容
        template.system_prompt = version.system_prompt
        template.user_prompt = version.user_prompt
        template.variables = version.variables
        template.model = version.model_config.get("model", template.model)
        template.temperature = version.model_config.get("temperature", template.temperature)
        template.max_tokens = version.model_config.get("max_tokens", template.max_tokens)

        # 提交为新版本（回滚也是一个版本）
        await self.commit(
            prompt_id, template,
            change_summary=f"回滚到 v{target_version}",
            author="system",
        )

        return template
```
